//! Behavioral Hash (BH) — TRION L0
//! sense  = SHA3-256(input || 0x00)
//! antisense = SHA3-256(input || 0xFF) XOR complement_transform(sense)
//! Verification: sense XOR antisense == complement_transform(sense)

use sha3::{Digest, Sha3_256};
use serde::{Deserialize, Serialize};
use chrono::{DateTime, Utc};
use anyhow::{Result, anyhow};
use crate::event_types::EventType;

fn complement_transform(data: &[u8; 32]) -> [u8; 32] {
    let mut out = [0u8; 32];
    for (i, b) in data.iter().enumerate() {
        out[i] = !b;
    }
    out
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct HashDNA {
    pub sense: [u8; 32],
    pub antisense: [u8; 32],
}

impl HashDNA {
    pub fn new(input: &[u8]) -> Self {
        let mut sense_hasher = Sha3_256::new();
        sense_hasher.update(input);
        sense_hasher.update(&[0x00u8]);
        let sense: [u8; 32] = sense_hasher.finalize().into();

        let mut antisense_hasher = Sha3_256::new();
        antisense_hasher.update(input);
        antisense_hasher.update(&[0xFFu8]);
        let antisense_raw: [u8; 32] = antisense_hasher.finalize().into();

        let complement = complement_transform(&sense);
        let mut antisense = [0u8; 32];
        for i in 0..32 {
            antisense[i] = antisense_raw[i] ^ complement[i];
        }

        Self { sense, antisense }
    }

    pub fn verify(&self) -> bool {
        let expected_complement = complement_transform(&self.sense);
        let mut xor_result = [0u8; 32];
        for i in 0..32 {
            xor_result[i] = self.sense[i] ^ self.antisense[i];
        }
        xor_result == expected_complement
    }

    pub fn sense_hex(&self) -> String    { hex::encode(self.sense) }
    pub fn antisense_hex(&self) -> String { hex::encode(self.antisense) }
}

/// magnitude_normalized = log10(usd_value + 1) / log10(max_observed_90d + 1)
pub fn normalize_magnitude(usd_value: f64, max_observed_90d: f64) -> f64 {
    let numerator   = (usd_value + 1.0_f64).log10();
    let denominator = (max_observed_90d + 1.0_f64).log10();
    if denominator == 0.0 { return 0.0; }
    (numerator / denominator).clamp(0.0, 1.0)
}

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct BehavioralHash {
    pub entity_id:            String,
    pub event_type:           EventType,
    pub magnitude_normalized: f64,
    pub context:              String,
    pub timestamp:            DateTime<Utc>,
    pub chain_id:             u64,
    pub block_hash:           String,
    pub hash_dna:             HashDNA,
}

impl BehavioralHash {
    pub fn new(
        entity_id:        String,
        event_type:       EventType,
        usd_value:        f64,
        max_observed_90d: f64,
        context:          String,
        timestamp:        DateTime<Utc>,
        chain_id:         u64,
        block_hash:       String,
    ) -> Result<Self> {
        if usd_value < 0.0        { return Err(anyhow!("USD value cannot be negative")); }
        if max_observed_90d <= 0.0 { return Err(anyhow!("max_observed_90d must be > 0")); }

        let magnitude_normalized = normalize_magnitude(usd_value, max_observed_90d);

        let preimage = format!(
            "{}|{}|{:.6}|{}|{}|{}|{}",
            entity_id,
            event_type.type_byte(),
            magnitude_normalized,
            context,
            timestamp.timestamp_nanos_opt().unwrap_or(0),
            chain_id,
            block_hash,
        );

        let hash_dna = HashDNA::new(preimage.as_bytes());

        Ok(Self {
            entity_id, event_type, magnitude_normalized,
            context, timestamp, chain_id, block_hash, hash_dna,
        })
    }

    pub fn verify(&self) -> bool {
        self.hash_dna.verify()
            && self.magnitude_normalized >= 0.0
            && self.magnitude_normalized <= 1.0
    }
}

//! Entity Resolution Protocol (ERP) — TRION L0
//! BEO_confidence = weighted(CF, ST, SC, BP)
//! Valid only when BEO_confidence > 0.75

use serde::{Deserialize, Serialize};
use anyhow::Result;
use std::collections::HashMap;

const W_CF:  f64 = 0.40;
const W_ST:  f64 = 0.25;
const W_SC:  f64 = 0.25;
const W_BP:  f64 = 0.10;
const W_SUM: f64 = W_CF + W_ST + W_SC + W_BP;

const BEO_THRESHOLD:        f64 = 0.75;
const RHO_TIMING:           f64 = 0.85;
const GRAPH_SIM_THRESHOLD:  f64 = 0.80;

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct AddressCluster {
    pub addresses:  Vec<String>,
    pub confidence: f64,
    pub cf_score:   f64,
    pub st_score:   f64,
    pub sc_score:   f64,
    pub bp_score:   f64,
    pub is_valid:   bool,
    pub entity_id:  Option<String>,
}

impl AddressCluster {
    pub fn compute(
        addresses: Vec<String>,
        cf_score:  f64,
        st_score:  f64,
        sc_score:  f64,
        bp_score:  f64,
    ) -> Self {
        let confidence = (W_CF * cf_score + W_ST * st_score
                        + W_SC * sc_score + W_BP * bp_score)
                        / W_SUM;
        let confidence = confidence.clamp(0.0, 1.0);
        let is_valid   = confidence > BEO_THRESHOLD;

        let entity_id = if is_valid {
            let input = addresses.join("|");
            let hash  = sha3_256_hex(input.as_bytes());
            Some(format!("BEO_{}", &hash[..16]))
        } else {
            None
        };

        Self { addresses, confidence, cf_score, st_score,
               sc_score, bp_score, is_valid, entity_id }
    }

    pub fn compute_st_score(correlation: f64) -> f64 {
        if correlation > RHO_TIMING { correlation } else { correlation * 0.5 }
            .clamp(0.0, 1.0)
    }

    pub fn compute_bp_score(graph_similarity: f64) -> f64 {
        if graph_similarity > GRAPH_SIM_THRESHOLD { graph_similarity } else { 0.0 }
    }
}

fn sha3_256_hex(input: &[u8]) -> String {
    use sha3::{Digest, Sha3_256};
    let mut hasher = Sha3_256::new();
    hasher.update(input);
    hex::encode(hasher.finalize())
}

pub fn detect_manipulation(inp: &ManipulationInput) -> ManipulationResult {
    let mut scores: Vec<(ManipulationType, f64)> = Vec::new();

    // TYPE 1: WASH_TRADING
    if inp.cyclic_flow_ratio > 0.60 && inp.counterparty_count < 5 {
        scores.push((ManipulationType::WashTrading, 0.70 * inp.cyclic_flow_ratio));
    }

    // TYPE 2: COORDINATED_PUMP
    if inp.sync_buy_ratio > 0.80 && inp.coordinated_beo_count >= 3 {
        scores.push((ManipulationType::CoordinatedPump, 0.85 * inp.sync_buy_ratio));
    }

    // TYPE 3: ORACLE_ATTACK — automatic 1.0
    if inp.ref_price_deviation_pct > 15.0 && inp.large_swap_within_10_blocks {
        scores.push((ManipulationType::OracleAttackAttempt, 1.0));
    }

    // TYPE 4: SYBIL_LIQUIDITY
    if inp.top5_lp_share > 0.80 && inp.funding_source_count < 3 {
        let concentration = 1.0 / inp.funding_source_count.max(1) as f64;
        scores.push((ManipulationType::SybilLiquidity, 0.60 * concentration));
    }

    // TYPE 5: GOVERNANCE_CAPTURE
    if inp.vote_hhi > 4000.0 && inp.proposal_age_hours < 48.0 {
        let hhi_norm    = ((inp.vote_hhi - 4000.0) / 6000.0).clamp(0.0, 1.0);
        let recency     = 1.0 - (inp.proposal_age_hours / 48.0);
        scores.push((ManipulationType::GovernanceCapture,
                     (0.75 * hhi_norm * recency).clamp(0.0, 1.0)));
    }

    // TYPE 6: MEV_EXTRACTION_SUSTAINED
    if inp.mev_rate_pct > 0.5 && inp.mev_days_sustained > 7 {
        let rate_f     = (inp.mev_rate_pct / 5.0).clamp(0.0, 1.0);
        let dur_f      = ((inp.mev_days_sustained - 7) as f64 / 30.0).clamp(0.0, 1.0);
        scores.push((ManipulationType::MevExtractionSustained,
                     (0.65 * rate_f * (1.0 + dur_f)).clamp(0.0, 1.0)));
    }

    // TYPE 7: FAKE_VOLUME_PROTOCOL
    if inp.volume_entropy < 0.30 && inp.volume_spike_multiplier > 10.0 {
        let spike = ((inp.volume_spike_multiplier - 10.0) / 90.0).clamp(0.0, 1.0);
        scores.push((ManipulationType::FakeVolumeProtocol,
                     (0.80 * (1.0 - inp.volume_entropy) * (1.0 + spike)).clamp(0.0, 1.0)));

    if scores.is_empty() {
        return ManipulationResult { mf_score: 0.0, active_types: vec![], dominant_type: None };

    let mf_score = scores.iter()

        .map(|(t, _)| t.clone());

    let active = scores.into_iter().map(|(t, _)| t).collect();
    ManipulationResult { mf_score, active_types: active, dominant_type: dominant }
}

//! Physical Richness Score Φ — TRION L1
//! Φ(t) = (1/N) · Σ [ w_i · H(f_i(t)) ]
//! Φ_adj = Φ · (1 - MF_score)

use super::features::PhysicalFeatures;
use super::manipulation::{detect_manipulation, ManipulationInput, ManipulationResult};
use serde::{Deserialize, Serialize};

#[derive(Debug, Clone, Serialize, Deserialize)]
pub struct PhiOutput {
    pub phi_raw:               f64,
    pub phi_adj:               f64,
    pub manipulation:          ManipulationResult,
    pub feature_contributions: [f64; 9],
}

pub fn compute_phi(
    features:     &PhysicalFeatures,
    manip_input:  &ManipulationInput,
) -> PhiOutput {
    let n = 9.0_f64;
    let feature_vals = features.as_array();

    let mut weighted_sum  = 0.0_f64;
    let mut contributions = [0.0_f64; 9];

    for i in 0..9 {
        let c = features.weights[i] * feature_vals[i];
        contributions[i] = c;
        weighted_sum     += c;
    }

    let phi_raw    = (weighted_sum / n).clamp(0.0, 1.0);
    let manip      = detect_manipulation(manip_input);
    let phi_adj    = (phi_raw * (1.0 - manip.mf_score)).clamp(0.0, 1.0);

    PhiOutput { phi_raw, phi_adj, manipulation: manip, feature_contributions: contributions }
}

/// TC(t) = 1 - max_lag / TTL_min
/// Valid iff TC > tc_minimum
pub fn temporal_coherence(
    plane_timestamps_ms: &[i64],
    reference_ms:        i64,
    ttl_min_ms:          i64,
    tc_minimum:          f64,
) -> (f64, bool) {
    if plane_timestamps_ms.is_empty() { return (0.0, false); }
    let max_lag = plane_timestamps_ms.iter()
        .map(|&t| (t - reference_ms).abs())
        .max()
        .unwrap_or(0) as f64;
    let tc = (1.0 - max_lag / ttl_min_ms as f64).clamp(0.0, 1.0);
    (tc, tc > tc_minimum)
}

"""
Physical Feature Extractor — TRION L1
Computes 9 behavioral features per asset per window.
All features normalized to [0,1].
"""
import math
from typing import List, Dict
from dataclasses import dataclass


def shannon_entropy_normalized(values: List[float]) -> float:
    total = sum(values)
    if total == 0:
        return 0.0
    probs  = [v / total for v in values if v > 0]
    raw_h  = -sum(p * math.log(p) for p in probs)
    max_h  = math.log(len(values)) if len(values) > 1 else 1.0
    return raw_h / max_h if max_h > 0 else 0.0


@dataclass
class FeatureWindow:
    volumes:           List[float]
    counterparties:    List[str]
    timestamps:        List[float]
    contracts_touched: List[str]
    inflow:            float
    outflow:           float
    protocol_ids:      List[str]
    gas_prices:        List[float]
    mev_tx_count:      int
    total_tx_count:    int


def extract_features(window: FeatureWindow) -> Dict[str, float]:
    if window.total_tx_count == 0:
        return {f"f{i}": 0.0 for i in range(1, 10)}

    # f1: Transaction volume entropy
    f1 = shannon_entropy_normalized(window.volumes) if window.volumes else 0.0

    # f2: Counterparty diversity
    unique_cp = len(set(window.counterparties))
    f2 = unique_cp / max(len(window.counterparties), 1)

    # f3: Temporal spacing entropy
    if len(window.timestamps) > 1:
        gaps = [max(0.0, window.timestamps[i+1] - window.timestamps[i])
                for i in range(len(window.timestamps) - 1)]
        f3 = shannon_entropy_normalized(gaps) if any(g > 0 for g in gaps) else 0.0
    else:
        f3 = 0.0

    # f4: Smart contract interaction
    unique_contracts = len(set(window.contracts_touched))
    f4 = min(unique_contracts / max(window.total_tx_count, 1), 1.0)

    # f5: Value flow directionality
    gross = window.inflow + window.outflow
    f5 = (1.0 - abs(window.inflow - window.outflow) / gross) if gross > 0 else 0.0

    # f6: Wallet architecture diversity (prefix entropy proxy)
    prefixes = [cp[:4] for cp in window.counterparties if len(cp) >= 4]
    counts   = {}
    for p in prefixes:
        counts[p] = counts.get(p, 0) + 1
    f6 = shannon_entropy_normalized(list(counts.values())) if counts else 0.0

    # f7: Cross-protocol breadth
    f7 = min(len(set(window.protocol_ids)) / 10.0, 1.0)

    # f8: Gas usage pattern entropy
    f8 = shannon_entropy_normalized(window.gas_prices) if window.gas_prices else 0.0

    # f9: MEV interaction (inverted — high MEV = lower f9)
    f9 = 1.0 - min(window.mev_tx_count / max(window.total_tx_count, 1), 1.0)

    result = {
        "f1": round(f1, 6), "f2": round(f2, 6), "f3": round(f3, 6),
        "f4": round(f4, 6), "f5": round(f5, 6), "f6": round(f6, 6),
        "f7": round(f7, 6), "f8": round(f8, 6), "f9": round(f9, 6),
    }

    for k, v in result.items():
        assert 0.0 <= v <= 1.0, f"Feature {k} out of [0,1]: {v}"

    return result

    dormancy_type: DormancyType
    dormancy_days: float         # T: days dormant

    team_continuity: float       # 0-1: same team signals

    T = inp.dormancy_days

    decay_factor = math.exp(-kappa * T)

    # Behavioral continuity: cosine similarity of pre and post vectors
    def cosine_sim(a: np.ndarray, b: np.ndarray) -> float:
        na = np.linalg.norm(a)
        nb = np.linalg.norm(b)
        if na == 0 or nb == 0:
            return 0.0
        return float(np.dot(a, b) / (na * nb))

    behavioral_continuity = cosine_sim(
        inp.pre_dormancy_vector, inp.reactivation_vector
    )

    # Cross-chain evidence g(C)
    cross_chain = max(0.0, min(1.0, inp.cross_chain_evidence))

    # Resurrection score
    delta = (W_D * decay_factor * W_C * behavioral_continuity * W_X * cross_chain)
    delta = max(0.0, min(1.0, delta * 3.0))  # scale to [0,1]

    # Classification logic
    outcome = classify_resurrection(
        delta, inp, behavioral_continuity, decay_factor
    )

        "delta_resurrection": round(delta, 6),
        "decay_factor": round(decay_factor, 6),
        "behavioral_continuity": round(behavioral_continuity, 6),
        "cross_chain_evidence": round(cross_chain, 6),
        "dormancy_type": inp.dormancy_type.value,
        "dormancy_days": T,
        "outcome": outcome.value,
        "kappa_used": kappa,

        return ResurrectionOutcome.GENUINE_CONTINUATION

    # Default: new entity in old shell
    return ResurrectionOutcome.NEW_ENTITY_OLD_SHELL

"""
Mental Confidence Score M(t) — TRION L3
M(t) = 1 - (PI_t / PI_baseline)
PI_t = current prediction interval width (conformal prediction)
CRITICAL: Confidence ≠ Accuracy. Tracked separately.
M_adj(t) = M_base(t) · (1 - OE_factor(t))
"""
import math
from dataclasses import dataclass, field
from typing import Optional, List
from collections import deque
import numpy as np


@dataclass
class ConformalPredictor:
    """
    Distribution-free conformal prediction intervals.
    Guarantees: 95% CI brackets true outcome 95% ± 2% over 90d window.
    """
    # Calibration residuals from held-out set
    residuals: List[float] = field(default_factory=list)
    alpha: float = 0.05  # 95% coverage target

    def calibrate(self, y_true: np.ndarray, y_pred: np.ndarray):
        """Update calibration set with new predictions vs outcomes."""
        self.residuals = sorted(abs(y_true - y_pred).tolist())

    def predict_interval(self, y_hat: float) -> tuple:
        """
        Compute CI_95 for a point prediction.
        Width = 2 * quantile(residuals, 1-α)
        CRITICAL: CI_95 is ALWAYS non-null — enforced here, compile-time in prod.
        """
        if not self.residuals:
            # Before calibration: use conservative interval
            margin = abs(y_hat) * 0.50 + 0.10
            return (y_hat - margin, y_hat + margin)

        n = len(self.residuals)
        idx = math.ceil((1 - self.alpha) * (n + 1)) - 1
        idx = max(0, min(idx, n - 1))
        margin = self.residuals[idx]
        return (y_hat - margin, y_hat + margin)

    def interval_width(self, y_hat: float) -> float:
        lo, hi = self.predict_interval(y_hat)
        return hi - lo

    def coverage_rate(self, y_true: np.ndarray, y_pred: np.ndarray) -> float:
        """Verify 95% CI is properly calibrated over test set."""
        covered = 0
        for yt, yp in zip(y_true, y_pred):
            lo, hi = self.predict_interval(yp)
            if lo <= yt <= hi:
                covered += 1
        return covered / len(y_true)


@dataclass
class MentalScoreState:
    """Per-asset mental confidence tracking."""
    asset_id: str
    predictor: ConformalPredictor = field(default_factory=ConformalPredictor)

    # Rolling accuracy history (90-day)
    accuracy_history: deque = field(default_factory=lambda: deque(maxlen=90))

    # Observer effect: rolling correlation (signal_t-1, behavioral_change_t)
    oe_history: deque = field(default_factory=lambda: deque(maxlen=30))

    # Baseline PI width (calibrated from first 30 days)
    pi_baseline: Optional[float] = None

    def compute_m_base(self, current_prediction: float) -> float:
        """
        M(t) = 1 - (PI_t / PI_baseline)
        If PI_baseline not set yet: return 0.5 (uniform prior)
        """
        pi_t = self.predictor.interval_width(current_prediction)

        if self.pi_baseline is None:
            if len(self.accuracy_history) >= 30:
                # Set baseline from first 30 days
                self.pi_baseline = pi_t
            return 0.50  # prior before calibration

        if self.pi_baseline == 0:
            return 0.0

        m = 1.0 - (pi_t / self.pi_baseline)
        return max(0.0, min(1.0, m))

    def compute_oe_factor(self, signal_strength: float, behavioral_change: float):
        """
        OE_factor = corr(signal_strength_{t-1}, behavioral_change_t)
        Measures how much TRION's own signals move the market.
        High OE → M_adj discounted.
        """
        self.oe_history.append((signal_strength, behavioral_change))
        if len(self.oe_history) < 10:
            return 0.0  # not enough data yet

        signals = [h[0] for h in self.oe_history]
        changes = [h[1] for h in self.oe_history]

        if np.std(signals) == 0 or np.std(changes) == 0:
            return 0.0

        corr = float(np.corrcoef(signals, changes)[0, 1])
        return max(0.0, corr)  # only positive correlation is observer effect

    def m_adjusted(
        self, current_prediction: float,
        signal_strength: float = 0.0,
        behavioral_change: float = 0.0,
    ) -> dict:
        """
        M_adj(t) = M_base(t) · (1 - OE_factor(t))
        Returns full M output struct.
        """
        m_base = self.compute_m_base(current_prediction)
        oe_factor = self.compute_oe_factor(signal_strength, behavioral_change)
        m_adj = m_base * (1.0 - oe_factor)

        ci_lo, ci_hi = self.predictor.predict_interval(current_prediction)

        return {
            "m_base": round(m_base, 6),
            "m_adj": round(max(0.0, min(1.0, m_adj)), 6),
            "oe_factor": round(oe_factor, 6),
            "oe_flag": oe_factor > 0.30,
            "pi_t": round(self.predictor.interval_width(current_prediction), 6),
            "pi_baseline": round(self.pi_baseline, 6) if self.pi_baseline else None,
            "ci_95": (round(ci_lo, 6), round(ci_hi, 6)),  # ALWAYS non-null
            "confidence_is_not_accuracy": True,  # explicit field per whitepaper
        }

"""
Source Credibility Evolution — TRION L3
CRED(source, t) = CRED(source, t-1) · α_decay + verification_events · β_update
α_decay = 0.99/day  — sources must earn trust continuously
Events: +1.0 verified, -2.0 falsified, -3.0 manipulation, -5.0 conflict

class CredibilityEvent(Enum):
    VERIFIED            = +1.0
    FALSIFIED           = -2.0
    MANIPULATION_DETECTED = -3.0
    CONFLICT_OF_INTEREST  = -5.0


ALPHA_DECAY  = 0.99   # daily decay rate
BETA_UPDATE  = 0.10   # learning rate for events
CRED_FLAG_THRESHOLD    = 0.30
CRED_EXCLUDE_THRESHOLD = 0.10
CRED_INITIAL           = 0.50  # neutral starting credibility


@dataclass
class SourceCredibility:
    source_id: str

    last_updated: Optional[datetime] = None
    flagged: bool = False
    excluded: bool = False
    event_log: List[dict] = field(default_factory=list)

    def apply_daily_decay(self, days_elapsed: float = 1.0):
        """Apply continuous credibility decay: α_decay^days."""
        self.cred = self.cred * (ALPHA_DECAY ** days_elapsed)
        self._update_status()

    def apply_event(self, event: CredibilityEvent, evidence: str = ""):
        """Apply a credibility event (verification or violation)."""
        delta = event.value * BETA_UPDATE
        self.cred = max(0.0, min(1.0, self.cred + delta))
        self.event_log.append({
            "event": event.name,
            "delta": delta,
            "cred_after": round(self.cred, 4),
            "ts": datetime.now(timezone.utc).isoformat(),
            "evidence": evidence,
        })
        self._update_status()

    def _update_status(self):
        self.flagged  = self.cred < CRED_FLAG_THRESHOLD
        self.excluded = self.cred < CRED_EXCLUDE_THRESHOLD

    def is_usable(self) -> bool:
        return not self.excluded

    def weighted_contribution(self, raw_value: float) -> float:
        """Returns credibility-weighted value for CA(t) computation."""
        if self.excluded:
            return 0.0
        return self.cred * raw_value


class CredibilityRegistry:
    """Registry of all ANIMA data sources with live CRED scores."""
    def __init__(self):
        self.sources: Dict[str, SourceCredibility] = {}

    def register(self, source_id: str) -> SourceCredibility:
        if source_id not in self.sources:
            self.sources[source_id] = SourceCredibility(source_id=source_id)
        return self.sources[source_id]

    def get(self, source_id: str) -> Optional[SourceCredibility]:
        return self.sources.get(source_id)

    def apply_daily_decay_all(self):
        for src in self.sources.values():
            src.apply_daily_decay(1.0)

    def usable_sources(self) -> List[SourceCredibility]:
        return [s for s in self.sources.values() if s.is_usable()]

    def cross_source_agreement(self, source_values: Dict[str, float]) -> float:
        """
        CA(t) = Σ_s CRED(s,t) · agreement(s,t) / Σ_s CRED(s,t)
        agreement(s,t) = 1 - |value_s - median_value| / range
        """
        usable = {sid: v for sid, v in source_values.items()
                  if sid in self.sources and self.sources[sid].is_usable()}
        if not usable:
            return 0.0

        values = list(usable.values())
        median = sorted(values)[len(values) // 2]
        val_range = max(values) - min(values) if len(values) > 1 else 1.0
        if val_range == 0:
            val_range = 1.0

        weighted_sum = 0.0
        cred_sum = 0.0
        for sid, v in usable.items():
            cred = self.sources[sid].cred
            agreement = 1.0 - abs(v - median) / val_range
            weighted_sum += cred * agreement
            cred_sum += cred

        return (weighted_sum / cred_sum) if cred_sum > 0 else 0.0

def compute_c_score(
    phi_adj: float,
    m_adj: float,
    sigma: float,
    k_score: float = 0.0,   # Phase 8
    a_score: float = 0.0,   # Phase 7
    asset_type: str = "MATURE_PROTOCOL",
    mode: str = "BALANCED",
) -> Tuple[float, dict]:
    """
    C(t) = α·Φ_adj + β·M_adj + γ·Σ + δ·K + ε·A
    Weights from asset_type profile, normalized to active planes.
    """
    weights = dict(WEIGHT_PROFILES.get(asset_type, WEIGHT_PROFILES["MATURE_PROTOCOL"]))

    # Zero out inactive planes and renormalize
    active = {}
    if phi_adj > 0 or True:  active["alpha"] = weights["alpha"]
    if m_adj > 0 or True:    active["beta"]  = weights["beta"]
    if sigma > 0 or True:    active["gamma"] = weights["gamma"]
    # K and A only included when live (non-zero weights only in Phase 8/7)
    if k_score > 0: active["delta"]   = weights["delta"]
    if a_score > 0: active["epsilon"] = weights["epsilon"]

    total_weight = sum(active.values())
    if total_weight == 0:
        return 0.0, weights

    # Normalize to active planes
    norm_alpha   = weights["alpha"]   / total_weight
    norm_beta    = weights["beta"]    / total_weight
    norm_gamma   = weights["gamma"]   / total_weight
    norm_delta   = weights.get("delta",   0.0) / total_weight
    norm_epsilon = weights.get("epsilon", 0.0) / total_weight

    c = (norm_alpha * phi_adj + norm_beta * m_adj + norm_gamma * sigma
         + norm_delta * k_score + norm_epsilon * a_score)

    return round(max(0.0, min(1.0, c)), 6), weights

    """Θ(t) = Θ_min + (Θ_max - Θ_min) · V(t)"""
    return THETA_MIN + (THETA_MAX - THETA_MIN) * max(0.0, min(1.0, volatility))

"""
Five-Plane Full C(t) — TRION L9
C(t) = α·Φ_adj(t) + β·M_adj(t) + γ·Σ(t) + δ·K(t) + ε·A(t)
Emergence verification: C(t) accuracy > max(any single plane)
"""
from dataclasses import dataclass
from typing import Tuple, Optional
from coherence_score import WEIGHT_PROFILES, dynamic_threshold, SignalType, TRIONSignal


@dataclass
class FivePlaneOutput:
    c_score: float
    phi_adj: float
    m_adj: float
    sigma: float
    k_score: float
    a_score: float
    weights_used: dict
    asset_type: str
    theta: float
    ci_95: Tuple[float, float]
    conf_genesis: float
    tc_valid: bool
    # Emergence tracking
    max_single_plane: float = 0.0
    emergence_detected: bool = False

    def signal_type(self) -> SignalType:
        if self.c_score >= self.theta and self.tc_valid:
            return SignalType.VALUATION
        return SignalType.SILENCE


def compute_five_plane_c(
    phi_adj: float,
    m_adj: float,
    sigma: float,
    k_score: float,
    a_score: float,
    asset_type: str = "MATURE_PROTOCOL",
    volatility: float = 0.10,
    conf_genesis: float = 0.5,
    ci_95: Tuple[float, float] = (0.0, 1.0),
    tc_valid: bool = True,
) -> FivePlaneOutput:
    """
    Full 5-plane C(t) with all asset-type profiles.
    Checks emergence: C(t) > max single plane.
    """
    w = WEIGHT_PROFILES.get(asset_type, WEIGHT_PROFILES["MATURE_PROTOCOL"])

    # All planes active — no renormalization needed
    c = (w["alpha"]   * phi_adj
       + w["beta"]    * m_adj
       + w["gamma"]   * sigma
       + w["delta"]   * k_score
       + w["epsilon"] * a_score)
    c = round(max(0.0, min(1.0, c)), 6)

    theta = dynamic_threshold(volatility)
    max_plane = max(phi_adj, m_adj, sigma, k_score, a_score)
    emergence = c > max_plane

    return FivePlaneOutput(
        c_score=c,
        phi_adj=phi_adj, m_adj=m_adj, sigma=sigma,
        k_score=k_score, a_score=a_score,
        weights_used=w, asset_type=asset_type,
        theta=theta, ci_95=ci_95,
        conf_genesis=conf_genesis, tc_valid=tc_valid,
        max_single_plane=round(max_plane, 6),
        emergence_detected=emergence,
    )


def information_conservation_check(
    bh_generated: float,
    a_absorbed: float,
    s_emitted: float,
    e_lost: float,
    prev_i_trion: float,
) -> dict:
    """
    I_TRION(t) = BH_generated + A_absorbed - S_emitted - E_lost
    dI_TRION/dt >= 0 ALWAYS — Thermodynamic Information Conservation
    """
    i_trion = prev_i_trion + bh_generated + a_absorbed - s_emitted - e_lost
    di_dt = bh_generated + a_absorbed - s_emitted - e_lost
    conserved = di_dt >= 0
    return {
        "i_trion": round(i_trion, 6),
        "di_dt": round(di_dt, 6),
        "conserved": conserved,
        "violation": not conserved,
    }

        uint256 totalSupply_,
        address publicGoodPool_,
        address unknownUnknownReserve_

        publicGoodPool = publicGoodPool_;
        unknownUnknownReserve = unknownUnknownReserve_;

        emit Transfer(address(0), msg.sender, totalSupply_);

        uint256 burnAmount      = fee * CONSUMPTION_BURN_BPS / BPS_DENOMINATOR;
        uint256 publicGoodShare = fee * PUBLIC_GOOD_BPS / BPS_DENOMINATOR;
        uint256 protocolShare   = fee - burnAmount - publicGoodShare;

        balanceOf[msg.sender] -= fee;
        // Burn — remove from supply
        // Public good — mandatory transfer
        balanceOf[publicGoodPool] += publicGoodShare;
        emit Transfer(msg.sender, publicGoodPool, publicGoodShare);
        emit Transfer(msg.sender, address(0), burnAmount);
        emit SignalConsumed(msg.sender, burnAmount, publicGoodShare);
    }

    function stakeValidator(uint256 amount) external {
        require(amount >= MIN_VALIDATOR_STAKE, "Below minimum validator stake");
        require(balanceOf[msg.sender] >= amount);
        balanceOf[msg.sender] -= amount;
        validatorStake[msg.sender] += amount;
        emit ValidatorStaked(msg.sender, amount);

        balanceOf[msg.sender] -= value;

        emit Transfer(msg.sender, to, value);
        return true;
